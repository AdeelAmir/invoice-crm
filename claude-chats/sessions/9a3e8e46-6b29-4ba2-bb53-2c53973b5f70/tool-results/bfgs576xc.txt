import { z } from "zod";
import { requireApiKey } from "@/lib/api-key";
import { badRequest, serializeInvoice } from "@/lib/api";
import { dueInvoices } from "@/lib/invoices/service";
import { InvoiceStatus } from "@/lib/generated/prisma/enums";

// GET /api/invoices/due?from=<ISO UTC>&to=<ISO UTC>&status=PENDING,FAILED,PROCESSING
const querySchema = z.object({
  from: z.iso.datetime({ offset: true }),
  to: z.iso.datetime({ offset: true }),
  status: z
    .string()
    .default("PENDING,FAILED,PROCESSING")
    .transform((s) => s.split(",").map((x) => x.trim().toUpperCase()))
    .pipe(z.array(z.enum(Object.values(InvoiceStatus) as [InvoiceStatus, ...InvoiceStatus[]])).min(1)),
});

export async function GET(request: Request) {
  const denied = requireApiKey(request);
  if (denied) return denied;

  const params = Object.fromEntries(new URL(request.url).searchParams);
  const parsed = querySchema.safeParse(params);
  if (!parsed.success) {
    return badRequest("from and to must be ISO-8601 date-times; status a comma list of statuses", z.flattenError(parsed.error).fieldErrors);
  }
  const from = new Date(parsed.data.from);
  const to = new Date(parsed.data.to);
  if (from > to) return badRequest("from must not be after to");

  const invoices = await dueInvoices({ from, to, statuses: parsed.data.status });
  return NextResponse.json({
    from: from.toISOString(),
    to: to.toISOString(),
    statuses: parsed.data.status,
    count: invoices.length,
    invoices: invoices.map(serializeInvoice),
  });
}

// POST /api/invoices/{id}/claim: atomically PENDING -> PROCESSING.
// 200 when this caller now owns the invoice, 409 when it was already claimed.
export async function POST(request: Request, ctx: RouteContext<"/api/invoices/[id]/claim">) {
  const denied = requireApiKey(request);
  if (denied) return denied;
  const id = parseInvoiceId((await ctx.params).id);
  if (!id) return notFoundJson();

  const result = await apiClaim(id, "api");
  if (!result.found) return notFoundJson();
  revalidatePath("/dashboard/invoices", "layout");

  const invoice = await prisma.invoice.findUniqueOrThrow({
    where: { id },
    include: { client: true, lineItems: true, taxes: true },
  });
  if (!result.claimed) {
    return NextResponse.json(
      { error: "Invoice is not PENDING (already claimed or processed)", invoice: serializeInvoice(invoice) },
      { status: 409 },
    );
  }
  return NextResponse.json({ claimed: true, invoice: serializeInvoice(invoice) });
}

import { randomUUID } from "node:crypto";
import { prisma } from "@/lib/db";
import { invoiceConfig } from "@/lib/config";
import { toCents } from "@/lib/money";
import { GatewayError, createGatewayInvoice } from "@/lib/gateway/mock";
import { logStep } from "@/lib/invoices/audit";
import { nextOccurrence, ruleAllowsNext } from "@/lib/invoices/recurrence";
import { retryDelaySeconds } from "@/lib/invoices/transitions";
import type { ErrorType } from "@/lib/generated/prisma/enums";

// The documented invoicing flow, one invoice at a time:
//   1. claim it (PENDING -> PROCESSING, atomic; skip if the claim fails)
//   2. create it in the payment gateway, keyed on the CRM invoice id
//   3. mark it CREATED with the gateway invoice id
//   4. (send to the client: out of scope for the demo)
//   5. log every step
// then, for a recurring invoice, generate the next occurrence.
//
// After the claim, every write is guarded on (status = PROCESSING and
// processingStartedAt = our claim time). If a stuck invoice was reset and
// claimed again in the meantime, the stale run can no longer touch it.

export type ProcessOutcome =
  | { outcome: "created"; gatewayInvoiceId: string; replayed: boolean; nextInvoiceId: number | null }
  | {
      outcome: "failed";
      error: string;
      errorType: ErrorType;
      retryCount: number;
      retryLimitReached: boolean;
    }
  | { outcome: "skipped"; reason: string };

const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

export function invoiceNumberFor(id: number): string {
  return `INV-${String(id).padStart(5, "0")}`;
}

export async function claimInvoice(invoiceId: number, actor: string): Promise<Date | null> {
  const claimedAt = new Date();
  const { count } = await prisma.invoice.updateMany({
    where: { id: invoiceId, status: "PENDING" },
    data: { status: "PROCESSING", processingStartedAt: claimedAt },
  });
  await logStep({
    invoiceId,
    step: "CLAIM",
    result: count === 1 ? "SUCCESS" : "FAILURE",
    message:
      count === 1
        ? "Claimed: PENDING → PROCESSING"
        : "Claim refused: the invoice is not PENDING (already claimed or processed)",
    actor,
  });
  return count === 1 ? claimedAt : null;
}

type LoadedInvoice = NonNullable<Awaited<ReturnType<typeof loadInvoice>>>;

function loadInvoice(invoiceId: number) {
  return prisma.invoice.findUnique({
    where: { id: invoiceId },
    include: { client: true, lineItems: true, taxes: true, recurringRule: true },
  });
}

// Permanent problems: retrying cannot help until someone fixes the data.
function dataProblem(invoice: LoadedInvoice): string | null {
  if (!invoice.client.isActive) return "Client is deactivated";
  if (!EMAIL_RE.test(invoice.client.email)) return `Invalid client email "${invoice.client.email}"`;
  if (invoice.lineItems.length === 0) return "Invoice has no line items";
  if (toCents(invoice.total) <= 0) return "Invoice total must be greater than zero";
  if (!/^[A-Z]{3}$/.test(invoice.currency)) return `Invalid currency "${invoice.currency}"`;
  return null;
}

export async function processInvoice(invoiceId: number, actor: string): Promise<ProcessOutcome> {
  const exists = await prisma.invoice.findUnique({ where: { id: invoiceId }, select: { id: true } });
  if (!exists) return { outcome: "skipped", reason: "Invoice not found" };

  const claimedAt = await claimInvoice(invoiceId, actor);
  if (!claimedAt) {
    return { outcome: "skipped", reason: "Invoice is already being processed or is not pending" };
  }
  const owned = { id: invoiceId, status: "PROCESSING" as const, processingStartedAt: claimedAt };

  const invoice = (await loadInvoice(invoiceId))!;

  try {
    const problem = dataProblem(invoice);
    if (problem) throw new GatewayError(problem, false, "INVALID_DATA");

    const gateway = await createGatewayInvoice({
      idempotencyKey: `crm-invoice-${invoice.id}`,
      amountCents: toCents(invoice.total),
      currency: invoice.currency,
      simulateTimeout: invoice.simulateFailure === "TIMEOUT" && invoice.retryCount === 0,
    });
    await logStep({
      invoiceId,
      step: "GATEWAY_CREATE",
      result: "SUCCESS",
      message: gateway.replayed
        ? `Gateway returned the existing invoice ${gateway.gatewayId} (idempotent replay)`
        : `Gateway invoice ${gateway.gatewayId} created`,
      actor,
      metadata: { gatewayInvoiceId: gateway.gatewayId, replayed: gateway.replayed },
    });

    const { count } = await prisma.invoice.updateMany({
      where: owned,
      data: {
        status: "CREATED",
        gatewayInvoiceId: gateway.gatewayId,
        lastError: null,
        errorType: null,
        nextRetryAt: null,
        processingStartedAt: null,
      },
    });
    if (count !== 1) {
      await logStep({
        invoiceId,
        step: "STATUS_UPDATE",
        result: "FAILURE",
        message: "Status not updated: this run no longer owns the invoice",
        actor,
      });
      return { outcome: "skipped", reason: "Another run took over this invoice" };
    }
    await logStep({
      invoiceId,
      step: "STATUS_UPDATE",
      result: "SUCCESS",
      message: "PROCESSING → CREATED",
      actor,
    });

    const nextInvoiceId = await generateNextOccurrence(invoice, actor);
    return {
      outcome: "created",
      gatewayInvoiceId: gateway.gatewayId,
      replayed: gateway.replayed,
      nextInvoiceId,
    };
  } catch (err) {
    return recordFailure(invoice, owned, err, actor);
  }
}

async function recordFailure(
  invoice: LoadedInvoice,
  owned: { id: number; status: "PROCESSING"; processingStartedAt: Date },
  err: unknown,
  actor: string,
): Promise<ProcessOutcome> {
  const { maxRetries, retryBaseSeconds } = invoiceConfig();
  // Anything that is not a known gateway error (network, DB hiccup) is
  // treated as temporary.
  const failure =
    err instanceof GatewayError
      ? err
      : new GatewayError(err instanceof Error ? err.message : "Unexpected error", true, "UNEXPECTED");
  const errorType: ErrorType = failure.retryable ? "RETRYABLE" : "PERMANENT";
  const retryCount = invoice.retryCount + 1;
  const retryLimitReached = retryCount >= maxRetries;
  const nextRetryAt =
    failure.retryable && !retryLimitReached
      ? new Date(Date.now() + retryDelaySeconds(retryCount, retryBaseSeconds) * 1000)
      : null;

  await logStep({
    invoiceId: invoice.id,
    step: "GATEWAY_CREATE",
    result: "FAILURE",
    message: `${failure.message} [${failure.code}]`,
    actor,
    metadata: { code: failure.code, errorType },
  });

  const { count } = await prisma.invoice.updateMany({
    where: owned,
    data: {
      status: "FAILED",
      retryCount: { increment: 1 },
      lastError: failure.message,
      errorType,
      nextRetryAt,
      processingStartedAt: null,
    },
  });
  if (count !== 1) {
    return { outcome: "skipped", reason: "Another run took over this invoice" };
  }

  const notes = [`PROCESSING → FAILED (${errorType}, attempt ${retryCount} of ${maxRetries})`];
  if (errorType === "PERMANENT") notes.push("admin alert raised: the data must be fixed before a retry");
  if (retryLimitReached) notes.push("retry limit reached: developer alert raised");
  else if (nextRetryAt) notes.push(`next automatic retry after ${nextRetryAt.toISOString()}`);
  await logStep({
    invoiceId: invoice.id,
    step: "STATUS_UPDATE",
    result: "SUCCESS",
    message: notes.join("; "),
    actor,
    metadata: { retryCount, retryLimitReached, nextRetryAt: nextRetryAt?.toISOString() ?? null },
  });

  return { outcome: "failed", error: failure.message, errorType, retryCount, retryLimitReached };
}

function isUniqueViolation(err: unknown): boolean {
  return typeof err === "object" && err !== null && (err as { code?: string }).code === "P2002";
}

// After an occurrence reaches CREATED, add the next one from the recurrence
// rule. Safe to repeat: parentInvoiceId is unique, so an occurrence can
// only ever have one successor.
async function generateNextOccurrence(invoice: LoadedInvoice, actor: string): Promise<number | null> {
  if (!invoice.recurringRuleId) return null;

  const existing = await prisma.invoice.findUnique({
    where: { parentInvoiceId: invoice.id },
    select: { id: true },
  });
  if (existing) return existing.id;

  try {
    return await prisma.$transaction(async (tx) => {
      const rule = await tx.recurringRule.findUniqueOrThrow({ where: { id: invoice.recurringRuleId! } });
      const nextIssueAt = nextOccurrence(invoice.issueAt, rule.frequency, invoice.client.timezone);
      if (!ruleAllowsNext(rule, nextIssueAt)) {
        await logStep(
          {
            invoiceId: invoice.id,
            step: "RECURRING_NEXT",
            result: "SUCCESS",
            message: "Recurring schedule complete: no further occurrence",
            actor,
          },
          tx,
        );
        return null;
      }

      const termMs = invoice.dueAt.getTime() - invoice.issueAt.getTime();
      const created = await tx.invoice.create({
        data: {
          invoiceNumber: `TMP-${randomUUID()}`,
          clientId: invoice.clientId,
          currency: invoice.currency,
          subtotal: invoice.subtotal,
          taxTotal: invoice.taxTotal,
          total: invoice.total,
          issueAt: nextIssueAt,
          dueAt: new Date(nextIssueAt.getTime() + termMs),
          paymentTerms: invoice.paymentTerms,
          type: "RECURRING",
          status: "PENDING",
          recurringRuleId: rule.id,
          parentInvoiceId: invoice.id,
          lineItems: {
            create: invoice.lineItems.map((l) => ({
              description: l.description,
              quantity: l.quantity,
              unitPrice: l.unitPrice,
              lineTotal: l.lineTotal,
              position: l.position,
            })),
          },
          taxes: {
            create: invoice.taxes.map((t) => ({ name: t.name, rate: t.rate, amount: t.amount })),
          },
        },
      });
      const number = invoiceNumberFor(created.id);
      await tx.invoice.update({ where: { id: created.id }, data: { invoiceNumber: number } });
      await tx.recurringRule.update({
        where: { id: rule.id },
        data: { generatedCount: { increment: 1 } },
      });
      await logStep(
        {
          invoiceId: invoice.id,
          step: "RECURRING_NEXT",
          result: "SUCCESS",
          message: `Next occurrence ${number} generated (issue ${nextIssueAt.toISOString()})`,
          actor,
          metadata: { nextInvoiceId: created.id },
        },
        tx,
      );
      await logStep(
        {
          invoiceId: created.id,
          step: "RECURRING_NEXT",
          result: "SUCCESS",
          message: `Generated from ${invoice.invoiceNumber} by the ${rule.frequency.toLowerCase()} schedule`,
          actor,
          metadata: { parentInvoiceId: invoice.id },
        },
        tx,
      );
      return created.id;
    });
  } catch (err) {
    if (isUniqueViolation(err)) {
      const winner = await prisma.invoice.findUnique({
        where: { parentInvoiceId: invoice.id },
        select: { id: true },
      });
      return winner?.id ?? null;
    }
    // The invoice itself is CREATED; a failed rollover must not undo that.
    await logStep({
      invoiceId: invoice.id,
      step: "RECURRING_NEXT",
      result: "FAILURE",
      message: `Could not generate the next occurrence: ${err instanceof Error ? err.message : "unknown error"}`,
      actor,
    });
    return null;
  }
}

import type { RecurrenceFrequency } from "@/lib/generated/prisma/enums";
import { wallTimeIn, zonedTimeToUtc } from "@/lib/timezone";

const MONTHS: Record<Exclude<RecurrenceFrequency, "WEEKLY">, number> = {
  MONTHLY: 1,
  QUARTERLY: 3,
  YEARLY: 12,
};

function daysInMonth(year: number, month: number): number {
  return new Date(Date.UTC(year, month, 0)).getUTCDate();
}

// Next occurrence of `from`, stepped in the client's own timezone: 09:00 in
// Montreal stays 09:00 after a DST change, and Jan 31 + 1 month is Feb 28/29.
export function nextOccurrence(from: Date, frequency: RecurrenceFrequency, timeZone: string): Date {
  const w = wallTimeIn(from, timeZone);
  if (frequency === "WEEKLY") {
    const d = new Date(Date.UTC(w.year, w.month - 1, w.day + 7));
    return zonedTimeToUtc(
      { ...w, year: d.getUTCFullYear(), month: d.getUTCMonth() + 1, day: d.getUTCDate() },
      timeZone,
    );
  }
  const total = w.year * 12 + (w.month - 1) + MONTHS[frequency];
  const year = Math.floor(total / 12);
  const month = (total % 12) + 1;
  const day = Math.min(w.day, daysInMonth(year, month));
  return zonedTimeToUtc({ ...w, year, month, day }, timeZone);
}

export type RuleState = {
  endDate: Date | null;
  occurrenceCount: number | null;
  generatedCount: number;
};

// Whether the rule allows another invoice issued at `nextIssueAt`.
export function ruleAllowsNext(rule: RuleState, nextIssueAt: Date): boolean {
  if (rule.occurrenceCount !== null && rule.generatedCount >= rule.occurrenceCount) return false;
  if (rule.endDate !== null && nextIssueAt.getTime() > rule.endDate.getTime()) return false;
  return true;
}

import { prisma } from "@/lib/db";
import { invoiceConfig } from "@/lib/config";
import { logStep } from "@/lib/invoices/audit";
import { claimInvoice, processInvoice, type ProcessOutcome } from "@/lib/invoices/processor";
import { ADMIN_TRANSITIONS, SYSTEM_TRANSITIONS, canTransition } from "@/lib/invoices/transitions";
import type { Prisma } from "@/lib/generated/prisma/client";
import type { AuditResult, AuditStep, ErrorType, InvoiceStatus } from "@/lib/generated/prisma/enums";

// Invoice operations shared by the admin UI (server actions) and the
// CRM API. Every status change is a guarded update on the status we read,
// so two concurrent callers can never both win.

function stuckBefore(): Date {
  return new Date(Date.now() - invoiceConfig().processingTimeoutMinutes * 60_000);
}

export function isStuck(invoice: { status: InvoiceStatus; processingStartedAt: Date | null }): boolean {
  return (
    invoice.status === "PROCESSING" &&
    invoice.processingStartedAt !== null &&
    invoice.processingStartedAt < stuckBefore()
  );
}

export function canRetry(invoice: {
  status: InvoiceStatus;
  retryCount: number;
  processingStartedAt: Date | null;
}): boolean {
  if (isStuck(invoice)) return true;
  return invoice.status === "FAILED" && invoice.retryCount < invoiceConfig().maxRetries;
}

// Which buttons the invoice page offers for this invoice.
export function actionFlags(invoice: {
  status: InvoiceStatus;
  retryCount: number;
  processingStartedAt: Date | null;
}) {
  return {
    canRetry: canRetry(invoice),
    canGenerate: invoice.status === "PENDING",
    manualStatuses: ADMIN_TRANSITIONS[invoice.status].filter(
      (s): s is "PAID" | "CANCELLED" => s === "PAID" || s === "CANCELLED",
    ),
    stuck: isStuck(invoice),
  };
}

// Retry: put a FAILED (under the retry limit) or stuck PROCESSING invoice
// back to PENDING, then run the normal flow on it.
export async function retryInvoice(invoiceId: number, actor: string): Promise<ProcessOutcome> {
  const { maxRetries } = invoiceConfig();
  const before = await prisma.invoice.findUnique({ where: { id: invoiceId } });
  if (!before) return { outcome: "skipped", reason: "Invoice not found" };

  const { count } = await prisma.invoice.updateMany({
    where: {
      id: invoiceId,
      OR: [
        { status: "FAILED", retryCount: { lt: maxRetries } },
        { status: "PROCESSING", processingStartedAt: { lt: stuckBefore() } },
      ],
    },
    data: { status: "PENDING", processingStartedAt: null, nextRetryAt: null },
  });
  if (count === 0) {
    const reason =
      before.status === "FAILED" && before.retryCount >= maxRetries
        ? `Retry limit reached (${before.retryCount} of ${maxRetries})`
        : `Only failed or stuck invoices can be retried (status is ${before.status})`;
    return { outcome: "skipped", reason };
  }

  await logStep({
    invoiceId,
    step: "RETRY",
    result: "SUCCESS",
    message: `Retry requested: ${before.status} → PENDING (attempt ${before.retryCount + 1} of ${maxRetries})`,
    actor,
  });
  return processInvoice(invoiceId, actor);
}

// Generate now: process a PENDING invoice immediately instead of waiting
// for the next scheduled cycle. The claim inside makes it safe to repeat.
export async function generateNow(invoiceId: number, actor: string): Promise<ProcessOutcome> {
  const invoice = await prisma.invoice.findUnique({ where: { id: invoiceId }, select: { status: true } });
  if (!invoice) return { outcome: "skipped", reason: "Invoice not found" };
  if (invoice.status !== "PENDING") {
    return { outcome: "skipped", reason: `Only pending invoices can be generated (status is ${invoice.status})` };
  }
  await logStep({
    invoiceId,
    step: "GENERATE_NOW",
    result: "SUCCESS",
    message: "Generate now requested",
    actor,
  });
  return processInvoice(invoiceId, actor);
}

export type StatusChangeResult = { ok: true; from: InvoiceStatus; to: InvoiceStatus } | { ok: false; error: string };

export async function adminSetStatus(
  invoiceId: number,
  to: InvoiceStatus,
  actor: string,
): Promise<StatusChangeResult> {
  const invoice = await prisma.invoice.findUnique({ where: { id: invoiceId }, select: { status: true } });
  if (!invoice) return { ok: false, error: "Invoice not found" };
  const from = invoice.status;
  if (!canTransition(ADMIN_TRANSITIONS, from, to)) {
    return { ok: false, error: `Cannot change status from ${from} to ${to}` };
  }
  const { count } = await prisma.invoice.updateMany({
    where: { id: invoiceId, status: from },
    data: { status: to, ...(to === "CANCELLED" ? { nextRetryAt: null } : {}) },
  });
  if (count !== 1) return { ok: false, error: "The invoice changed meanwhile; reload and try again" };
  await logStep({
    invoiceId,
    step: "MANUAL_STATUS",
    result: "SUCCESS",
    message: `Status changed by admin: ${from} → ${to}`,
    actor,
  });
  return { ok: true, from, to };
}

  status: InvoiceStatus;
  gatewayInvoiceId?: string;
  error?: string;
  errorType?: ErrorType;

// PATCH /invoices/{id}/status, used by the automation.
export async function systemSetStatus(
  invoiceId: number,
  input: SystemStatusInput,
  actor: string,
): Promise<StatusChangeResult> {
  const invoice = await prisma.invoice.findUnique({
    where: { id: invoiceId },
    select: { status: true, gatewayInvoiceId: true },
  });
  if (!invoice) return { ok: false, error: "Invoice not found" };
  const from = invoice.status;
  const to = input.status;
  if (!canTransition(SYSTEM_TRANSITIONS, from, to)) {
    return { ok: false, error: `Cannot change status from ${from} to ${to}` };
  }
  if (to === "CREATED" && !input.gatewayInvoiceId && !invoice.gatewayInvoiceId) {
    return { ok: false, error: "gatewayInvoiceId is required to mark an invoice CREATED" };
  }
  if (to === "FAILED" && !input.error) {
    return { ok: false, error: "error is required to mark an invoice FAILED" };
  }

  const data: Prisma.InvoiceUpdateManyMutationInput = { status: to };
  if (input.gatewayInvoiceId) data.gatewayInvoiceId = input.gatewayInvoiceId;
  if (to === "FAILED") {
    data.retryCount = { increment: 1 };
    data.lastError = input.error;
    data.errorType = input.errorType ?? "RETRYABLE";
    data.processingStartedAt = null;
  }
  if (to === "CREATED" || to === "PENDING") {
    data.processingStartedAt = null;
  }
  if (to === "CREATED") {
    data.lastError = null;
    data.errorType = null;
    data.nextRetryAt = null;
  }

  try {
    const { count } = await prisma.invoice.updateMany({ where: { id: invoiceId, status: from }, data });
    if (count !== 1) return { ok: false, error: "The invoice changed meanwhile; read it again" };
  } catch (err) {
    if ((err as { code?: string }).code === "P2002") {
      return { ok: false, error: "gatewayInvoiceId is already used by another invoice" };
    }
    throw err;
  }
  await logStep({
    invoiceId,
    step: "STATUS_UPDATE",
    result: "SUCCESS",
    message: `${from} → ${to}${input.error ? `: ${input.error}` : ""}`,
    actor,
    metadata: { gatewayInvoiceId: input.gatewayInvoiceId ?? null, errorType: input.errorType ?? null },
  });
  return { ok: true, from, to };
}

// GET /invoices/due: what a processing cycle should pick up. An invoice is
// due once its issue time falls in the window. With the default statuses
// this also returns failed invoices that may be retried now and invoices
// stuck in PROCESSING longer than the timeout.
export async function dueInvoices(params: { from: Date; to: Date; statuses: InvoiceStatus[] }) {
  const { maxRetries } = invoiceConfig();
  const or: Prisma.InvoiceWhereInput[] = [];
  for (const status of params.statuses) {
    if (status === "FAILED") {
      or.push({
        status: "FAILED",
        retryCount: { lt: maxRetries },
        errorType: "RETRYABLE",
        OR: [{ nextRetryAt: null }, { nextRetryAt: { lte: params.to } }],
      });
    } else if (status === "PROCESSING") {
      or.push({ status: "PROCESSING", processingStartedAt: { lt: stuckBefore() } });
    } else {
      or.push({ status });
    }
  }
  return prisma.invoice.findMany({
    where: { issueAt: { gte: params.from, lte: params.to }, OR: or },
    orderBy: [{ issueAt: "asc" }, { id: "asc" }],
    include: { client: true, lineItems: true, taxes: true },
  });
}

import type { InvoiceStatus } from "@/lib/generated/prisma/enums";

type TransitionTable = Record<InvoiceStatus, readonly InvoiceStatus[]>;

// What the automation (agent / API) may move an invoice to. PENDING ->
// PROCESSING is missing on purpose: only the atomic claim may do that.
export const SYSTEM_TRANSITIONS: TransitionTable = {
  PENDING: ["CANCELLED"],
  PROCESSING: ["CREATED", "FAILED", "PENDING"],
  CREATED: ["SENT", "PAID", "FAILED", "CANCELLED"],
  SENT: ["PAID", "FAILED", "CANCELLED"],
  FAILED: ["PENDING", "CANCELLED"],
  PAID: [],
  CANCELLED: [],
};

// What an admin may set by hand on the invoice page. Retry and Generate
// now are separate actions that run the processing flow instead.
export const ADMIN_TRANSITIONS: TransitionTable = {
  PENDING: ["CANCELLED"],
  PROCESSING: [],
  CREATED: ["PAID", "CANCELLED"],
  SENT: ["PAID", "CANCELLED"],
  FAILED: ["CANCELLED"],
  PAID: [],
  CANCELLED: [],
};

export function canTransition(
  table: TransitionTable,
  from: InvoiceStatus,
  to: InvoiceStatus,
): boolean {
  return table[from].includes(to);
}

// Exponential backoff before the next automatic retry: base, 2x, 4x, ...
export function retryDelaySeconds(retryCount: number, baseSeconds: number): number {
  return baseSeconds * 2 ** Math.max(0, retryCount - 1);
}

// Stand-in for a real payment gateway (Stripe, Moneris, ...). It keeps the
// one property the flow depends on: creating an invoice is idempotent on
// the key, so a retry after a crash never bills the client twice.

export class GatewayError extends Error {
  constructor(
    message: string,
    readonly retryable: boolean,
    readonly code: string,
  ) {
    super(message);
    this.name = "GatewayError";
  }
}

export type CreateGatewayInvoiceInput = {
  idempotencyKey: string;
  amountCents: number;
  currency: string;
  simulateTimeout?: boolean;
};

export type GatewayInvoice = { gatewayId: string; replayed: boolean };

function isUniqueViolation(err: unknown): boolean {
  return typeof err === "object" && err !== null && (err as { code?: string }).code === "P2002";
}

export async function createGatewayInvoice(
  input: CreateGatewayInvoiceInput,
): Promise<GatewayInvoice> {
  const { gatewayLatencyMs } = invoiceConfig();
  if (gatewayLatencyMs > 0) await new Promise((r) => setTimeout(r, gatewayLatencyMs));

  if (input.simulateTimeout) {
    throw new GatewayError("Payment gateway timed out (simulated)", true, "TIMEOUT");
  }

  const existing = await prisma.mockGatewayInvoice.findUnique({
    where: { idempotencyKey: input.idempotencyKey },
  });
  if (existing) return { gatewayId: existing.gatewayId, replayed: true };

  try {
    const created = await prisma.mockGatewayInvoice.create({
      data: {
        idempotencyKey: input.idempotencyKey,
        gatewayId: `gw_in_${randomUUID().replace(/-/g, "").slice(0, 16)}`,
        amount: fromCents(input.amountCents),
        currency: input.currency,
      },
    });
    return { gatewayId: created.gatewayId, replayed: false };
  } catch (err) {
    // Two calls raced on the same key: the other one won, return its result.
    if (isUniqueViolation(err)) {
      const winner = await prisma.mockGatewayInvoice.findUniqueOrThrow({
        where: { idempotencyKey: input.idempotencyKey },
      });
      return { gatewayId: winner.gatewayId, replayed: true };
    }
    throw err;
  }
}

"use server";

import { revalidatePath } from "next/cache";
import { z } from "zod";
import { requireAdmin } from "@/lib/auth/session";
import { adminSetStatus, generateNow, retryInvoice } from "@/lib/invoices/service";
import type { ProcessOutcome } from "@/lib/invoices/processor";
import { STATUS_LABELS } from "@/lib/format";

export type ActionResult = { ok: boolean; message: string };

const idSchema = z.number().int().positive();
const statusSchema = z.enum(["CANCELLED", "PAID"]);

function refresh(id: number) {
  revalidatePath("/dashboard/invoices");
  revalidatePath(`/dashboard/invoices/${id}`);
  revalidatePath("/dashboard/clients", "layout");
}

function describe(result: ProcessOutcome): ActionResult {
  switch (result.outcome) {
    case "created":
      return {
        ok: true,
        message:
          `Invoice created in the payment gateway (${result.gatewayInvoiceId}).` +
          (result.nextInvoiceId ? " Next recurring occurrence generated." : ""),
      };
    case "failed":
      return {
        ok: false,
        message:
          `Processing failed: ${result.error}.` +
          (result.retryLimitReached ? " Retry limit reached." : ` (${result.errorType.toLowerCase()} error)`),
      };
    case "skipped":
      return { ok: false, message: result.reason };
  }
}

export async function retryInvoiceAction(invoiceId: number): Promise<ActionResult> {
  const admin = await requireAdmin();
  const id = idSchema.parse(invoiceId);
  const result = await retryInvoice(id, `admin:${admin.id}`);
  refresh(id);
  return describe(result);
}

export async function generateNowAction(invoiceId: number): Promise<ActionResult> {
  const admin = await requireAdmin();
  const id = idSchema.parse(invoiceId);
  const result = await generateNow(id, `admin:${admin.id}`);
  refresh(id);
  return describe(result);
}

export async function setStatusAction(invoiceId: number, status: string): Promise<ActionResult> {
  const admin = await requireAdmin();
  const id = idSchema.parse(invoiceId);
  const parsed = statusSchema.safeParse(status);
  if (!parsed.success) return { ok: false, message: "That status cannot be set by hand" };
  const result = await adminSetStatus(id, parsed.data, `admin:${admin.id}`);
  refresh(id);
  return result.ok
    ? { ok: true, message: `Status changed: ${STATUS_LABELS[result.from]} → ${STATUS_LABELS[result.to]}` }
    : { ok: false, message: result.error };
}
